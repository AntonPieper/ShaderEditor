# The words of GLSL ES 1.00 and 3.00 a fragment shader can use, in
# ShaderEditor's own words, from the facts of the two specifications.
#
# One block per word, blocks apart by an empty line:
#   <word> <kind> <versions>  kind: function, type, qualifier, keyword,
#                             variable or constant; versions: 100, 300
#   form <declaration>        each form, as many as there are
#   detail <sentence>         about the forms: what their letters stand
#                             for, which forms are left out
#   text <sentence>           what it is or does; every block has one
#   note <sentence>           what else to know
#   example <code>            a line of an example, as many as needed

void type 100 300
text No value: the return type of a function that returns nothing.
example void main(void) {

bool type 100 300
form bool(x)
text true or false. Conditions of if, for, while and ?: must be bool.
example bool inside = length(p) < 0.5;

float type 100 300
form float(x)
text A floating-point number; highp, mediump or lowp, or the precision statement, set how exact it is.
note No number is converted by itself: write 1.0, not 1.
example float d = length(uv) * 2.0;

int type 100 300
form int(x)
text A signed whole number. int(x) drops the fraction of a float.
example int steps = int(t * 8.0);

uint type 300
form uint(x)
text An unsigned 32-bit whole number. Its literals end in u, as in 7u.
example uint h = uint(gl_FragCoord.x) * 747796405u;

true keyword 100 300
text The bool value true.

false keyword 100 300
text The bool value false.

vec2 type 100 300
form vec2(float x, float y)
form vec2(float v)
text Two floats, read as .x .y, .r .g or .s .t, in any order and repeated: p.yx, p.xx.
example vec2 uv = gl_FragCoord.xy / resolution;

form vec3(float x, float y, float z)

example gl_FragColor = vec4(color, 1.0);

bvec4 type 100 300

uvec4 type 300

mat4x4 type 300

samplerExternalOES type 100 300
text A texture the system fills, such as a camera frame; ShaderEditor's camera uniforms have this type.
note It needs #extension GL_OES_EGL_image_external : require in 1.00, and GL_OES_EGL_image_external_essl3 in 3.00.

gl_DepthRangeParameters type 100 300
form struct gl_DepthRangeParameters { float near; float far; float diff; }
text The type of gl_DepthRange: the near and far ends of the depth range and far - near.

struct keyword 100 300
text Declares a type of named fields; its name is then also its constructor.
example struct Ray { vec3 origin; vec3 dir; };

attribute qualifier 100
text An input of a vertex shader, one value per vertex. A fragment shader cannot use it.
note 3.00 reserves the word; a vertex shader there uses in.

varying qualifier 100
text A value the vertex shader writes and the fragment shader reads, blended across the triangle.
note 3.00 reserves the word: it is out in the vertex shader and in in the fragment shader.

centroid qualifier 300
text With in or out: with multisampling, the value is taken at a point inside the triangle's covered part of the pixel.

in qualifier 100 300
text A parameter copied into the function, the default. In 3.00 also an input of the fragment shader from the vertex shader.
example float shade(in vec3 n) {

out qualifier 100 300
text A parameter the function writes for the caller. In 3.00 also an output of the fragment shader, such as its colour.
example out vec4 fragColor;

inout qualifier 100 300
text A parameter copied in and written back to the caller at the end.
example void rotate(inout vec2 p, float a) {

uniform qualifier 100 300
text A value the app sets, the same for every pixel of a frame. ShaderEditor fills its own uniforms by name.
example uniform float time;

flat qualifier 300
text With in or out: no blending; the value of one vertex is used for the whole triangle. Integer inputs need it.

smooth qualifier 300
text With in or out: blended across the triangle, corrected for perspective; the default.

layout qualifier 300
text Sets where an input or output goes; layout(location = 1) writes an output to the second draw buffer.
example layout(location = 0) out vec4 fragColor;

invariant qualifier 100 300
text An output computed the same way in every program that writes it, so that equal inputs give equal results.

const qualifier 100 300
text A value fixed when the shader compiles, or a parameter the function does not change.
example const float PI = 3.14159265;

highp keyword 100 300
text The highest precision; in 3.00 floats and ints of 32 bits.
note A fragment shader in 1.00 may lack it; GL_FRAGMENT_PRECISION_HIGH is defined where it has it.
example #ifdef GL_FRAGMENT_PRECISION_HIGH

mediump keyword 100 300
text Medium precision: floats up to at least 16384 with about 3 decimal digits.
note Enough for colours and positions on screen; long times and large coordinates lose detail.

lowp keyword 100 300
text Low precision: floats at least from -2 to 2 in steps of 1/256, enough for a colour.

precision keyword 100 300
form precision <highp, mediump or lowp> <type>;
text Sets the precision of a type for every later declaration that names none.
note A fragment shader has no default for float: without this statement every float needs its own.
example precision mediump float;

while keyword 100 300
text Repeats the statement as long as the condition is true; the condition is checked first.
note A 1.00 GPU does not have to support while; for with a constant bound always works.

do keyword 100 300
text Runs the statement, then repeats it while the condition after while is true.
note A 1.00 GPU does not have to support do; for with a constant bound always works.
example do { p *= 0.5; } while (length(p) > 1.0);

for keyword 100 300
text Repeats the statement: for (start; condition; step).
note 1.00 only promises loops of one int or float counter, compared with a constant and changed by a constant.
example for (int i = 0; i < 8; i++) {

break keyword 100 300
text Leaves the innermost loop, or in 3.00 a switch.

continue keyword 100 300
text Skips the rest of this round of the innermost loop.

if keyword 100 300
text Runs the statement only when the bool condition is true.
example if (d < 0.0) {

else keyword 100 300
text What runs when the condition of the if before it is false.

discard keyword 100 300
text Drops the current pixel: nothing is written for it. Fragment shaders only.
example if (alpha < 0.5) discard;

return keyword 100 300
text Leaves the function, with the value it returns unless it is void.

switch keyword 300
text Jumps to the case label equal to an int or uint value, or to default.
note 1.00 reserves the word.
example switch (mode) { case 0: c = a; break; default: c = b; }

case keyword 300
text A label of a switch: where to go on for this value.

default keyword 300
text The label of a switch for every value no case names.

gl_FragCoord variable 100 300
form vec4 gl_FragCoord
text The pixel's place in the window: .xy in pixels, (0.5, 0.5) at the centre of the bottom-left pixel; .z its depth.
note It is mediump in 1.00 and highp in 3.00.
example vec2 uv = gl_FragCoord.xy / resolution.xy;

gl_FragColor variable 100

gl_FragData variable 100

gl_PointCoord variable 100 300

gl_DepthRange variable 100 300
