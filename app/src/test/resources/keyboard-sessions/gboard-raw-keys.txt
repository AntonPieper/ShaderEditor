# Gboard on the owner's phone with native suggestions hidden (the app's
# default, TYPE_NULL), as the prototype logged it: every key arrives as
# key events that the view handles, and the keyboard reads around the
# caret at focus and after each key. Written by hand from that log, on
# a small text. A key's up event may arrive before or after the view
# handled its down event.
start 15 15 "void main() {\n\t\n}"
getTextBeforeCursor 600 -> "void main() {\n\t"
getTextAfterCursor 600 -> "\n}"
getSelectedText -> null

# sendKeyEvent 1 34 -> true
app replace 15 15 "f"

getTextAfterCursor 600 -> "\n}"
getSelectedText -> null

app replace 16 16 "o"
updateSelection 17 17 -1 -1

getSelectedText -> null

# sendKeyEvent 1 66 -> true
